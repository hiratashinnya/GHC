2"""Targeted (single-file) dashboard update helpers. 

3 

4責務: 

5 AI が 1 ファイルを書き込んだ際に、対応するダッシュボード行のみを 

6 更新するための純粋ビルダーを提供する。全走査を避け最小限の

8 

9入力 / 出力: 

10 各関数の docstring を参照。

32def build_phase_row(phase, phase_idx, docs="docs"): 

33 """指定フェーズの 1 行のみを返す。 

34 

35 入力: 

36 phase (str) : フェーズ名 ("requirements" など)。 

37 phase_idx (int) : テーブルに表示する 1-based フェーズ番号。 

38 docs (str) : docs ルートディレクトリ (デフォルト: "docs")。 

39 

40 出力: 

41 str: テーブル行文字列 ("| フェーズN: ... | ... |")。 

42 """ 

43 pp = os.path.join(docs, phase) 

44 fmap = DD_OVERVIEW if phase == "detailed-design" else PROC_FILE 

45 cols = {str(p): _resolve_status_emoji(os.path.join(pp, fmap[p])) for p in range(1, 6)} 

46 cols["2v"] = ( 

47 _resolve_status_emoji(os.path.join(pp, DD_VALIDATION)) 

48 if phase == "detailed-design" 

49 else NOT_STARTED 

50 ) 

51 label = f"フェーズ{phase_idx}: {PHASE_LABEL.get(phase, phase)}" 

52 return ( 

53 f"| {label} | {cols['1']} | {cols['2']} | {cols['2v']}" 

54 f" | {cols['3']} | {cols['4']} | {cols['5']} |" 

55 )

84def _bottleneck_line_for(fp, fm): 

85 """フロントマター辞書からボトルネック行文字列を生成する。 

86 

87 責務: 

88 fmのstatus / approval-required を評価し、ボトルネック該当時は箇条文字列を返す。非該当時は None。 

89 

90 入力: 

91 fp (str) : ファイルパス文字列 (表示用)。 

92 fm (dict) : フロントマター辞書。 

93 

94 出力: 

95 str | None: ボトルネック行文字列。非該当時は None。 

96 """ 

97 st = fm.get("status", "") 

98 ar = fm.get("approval-required", False) 

99 if st == "rejected": 

100 return f"- `{fp}`: {st}（却下）" 

101 if st == "under-revision": 

102 return f"- `{fp}`: {st}（差し戻し影響）" 

103 if ar and st != "approved": 

104 return f"- `{fp}`: {st}（承認待ち）" 

105 return None

108def patch_bottleneck_line(dashboard_text, changed_file, docs="docs"): 

109 """ボトルネックセクション内の changed_file の行だけを更新する。 

110 

111 責務: 

112 1 ファイルの parse_fm のみ実行して新しいボトルネック行を決定し、 

113 ダッシュボードテキストのボトルネックセクションで 

114 changed_file に対応する既存行を 更新 / 追加 / 削除 する。 

115 ボトルネックセクションが「- なし」だけの場合も正しく処理する。 

116 

117 入力: 

118 dashboard_text (str) : dashboard.md の全テキスト。 

119 changed_file (str) : AI が書き込んだファイルパス。 

120 docs (str) : docs ルートディレクトリ (デフォルト: "docs")。 

121 

122 出力: 

123 str: 更新後の dashboard.md 全テキスト。ボトルネックセクションが 

124 なければ元のテキストをそのまま返す。 

125 """ 

126 rel = norm(changed_file) 

127 if not rel.startswith(norm(docs) + "/"): 

128 rel = norm(os.path.join(docs, changed_file)) if not os.path.isabs(changed_file) else rel 

129 

130 fm = parse_fm(changed_file) 

131 new_line = _bottleneck_line_for(rel, fm) if fm else None 

132 

133 sec_pat = re.compile( 

134 r"(## ボトルネック[^\n]*\n)(.*?)(?=\n## |\Z)", 

135 re.DOTALL, 

136 ) 

137 m = sec_pat.search(dashboard_text) 

138 if not m: 

139 return dashboard_text 

140 

141 body = m.group(2) 

142 escaped = re.escape(rel) 

143 existing_pat = re.compile(rf"^- `{escaped}`:.*$", re.MULTILINE) 

144 existing_match = existing_pat.search(body) 

145 

146 if existing_match: 

147 if new_line: 

148 body = existing_pat.sub(new_line, body) 

149 else: 

150 body = existing_pat.sub("", body) 

151 body = re.sub(r"\n{2,}", "\n", body).strip() 

152 if not body or re.fullmatch(r"[\s\n]*", body): 

153 body = "- なし" 

154 else: 

155 if new_line: 

156 if body.strip() == "- なし": 

157 body = new_line 

158 else: 

159 body = body.rstrip("\n") + "\n" + new_line 

160 

161 return dashboard_text[: m.start(2)] + body.rstrip("\n") + "\n" + dashboard_text[m.end(2):]
