1#!/usr/bin/env python3 

2"""Markdown text manipulation utilities. 

3 

4責務: 

5 ダッシュボード Markdown テキストに対するセクション置換・テーブル行置換の 

6 純粋関数を提供する。ファイル I/O は行わない。 

7 

8入力 / 出力: 

9 各関数の docstring を参照。 

10 

11副作用: 

12 なし (全関数純粋関数)。 

13 

14依存モジュール: 

15 - re (標準ライブラリ) 

16""" 

17 

18import re 

19 

20 

21def replace_section(text, heading_re, new_body): 

22 """## 見出しから次の ## または EOF までの内容を置換する。 

23 

24 入力: 

25 text (str) : 置換対象の Markdown 全体文字列。 

26 heading_re (str) : ## タイトルにマッチさせる正規表現パターン。 

27 new_body (str) : 見出し次のセクションに嵌め込む新しい本文。 

28 

29 出力: 

30 str: 置換後の Markdown 全体文字列。マッチしない場合は元の text。 

31 """ 

32 pat = re.compile( 

33 r"(## " + heading_re + r"[^\n]*\n)(.*?)(?=\n## |\Z)", 

34 re.DOTALL, 

35 ) 

36 m = pat.search(text) 

37 if m: 

38 return text[: m.start(2)] + new_body + "\n" + text[m.end(2):] 

39 return text

41 

42def replace_table_row(text, row_prefix_re, new_row): 

43 """テーブル内の特定行のみを置換する。 

44 

45 入力: 

46 text (str) : ダッシュボード全テキスト。 

47 row_prefix_re (str) : 行冒頭部分にマッチさせる正規表現パターン。 

48 new_row (str) : 置換後の行文字列 (改行なし)。 

49 

50 出力: 

51 str: 置換後のテキスト。マッチしない場合は元の text。 

52 """ 

53 return re.sub( 

54 r"^" + row_prefix_re + r".*$", 

55 new_row, 

56 text, 

57 count=1, 

58 flags=re.MULTILINE, 

59 )
