1#!/usr/bin/env python3 

2"""ac_rule_engine.py — Rule evaluation engine for the access control hook. 

3 

4責務: 

5 ツール呼び出しコンテキストとアクセス制御設定を照合し、 

6 適用すべきアクション（deny / confirm / allow）を決定する。 

7 

8入力 / 出力: 

9 evaluate(config: AccessControlConfig, context: MatchContext) -> Optional[RuleMatch] 

10 - マッチするルールがなければ None (= allow) 

11 - 複数マッチ時は deny > confirm の優先度で最上位のルールを返す 

12 - action="disabled" のルールは評価対象外 

13 

14副作用: 

15 なし（純粋関数群）。 

16 

17依存モジュール: 

18 fnmatch, dataclasses, pathlib, typing (標準ライブラリ) 

19 ac_config_loader (本プロジェクト) 

20 

21拡張ガイド: 

22 新しい操作タイプ（例: "network"）を追加する場合: 

23 1. NETWORK_TOOLS frozenset を追加する 

24 2. _determine_operation_type() に elif を追加する 

25 3. _get_candidate_rules() に elif を追加する 

26 4. _matches_rule() に対応する条件分岐を追加する 

27 5. ac_config_loader.py / access-control.json に network_rules を追加する 

28 

29 MatchContext に新フィールド（agent_id 等）を追加する場合: 

30 1. MatchContext dataclass にフィールドを追加する 

31 2. _matches_rule() 内で WhenClause の対応フィールドをチェックする 

32 （access_control.py 側でコンテキスト生成時にフィールドを渡す） 

33"""

157def _to_posix_relative(path_str: str, cwd: str) -> str: 

158 """パスを forward-slash・ワークスペース相対に正規化する。 

159 

160 絶対パスかつ cwd が判明している場合は cwd で相対化を試みる。 

161 相対化できない場合はそのまま forward-slash に変換して返す。 

162 

163 # Note: workspace_utils.to_workspace_relative への置換は不採用。 

164 # 理由: workspace 外の絶対パス（例: D:/other/... 形式）を評価対象として残すために 

165 # None を返す同関数へ置き換えてしまうと、ブロックルールが到達しなくなる。 

166 """ 

167 if not path_str: 

168 return "" 

169 path = Path(path_str) 

170 if path.is_absolute() and cwd: 

171 try: 

172 path = path.relative_to(cwd) 

173 except ValueError: 

174 pass 

175 return PurePosixPath(path).as_posix()

182def _match_single_path(normalized: str, pattern: str) -> bool: 

183 """1パスが1パターンにマッチするか判定する。 

184 

185 ``**/`` プレフィックスを持つパターンはルート直下のファイルにも適用する。 

186 例: ``**/.env`` は ``.env``（ルート直下）と ``subdir/.env`` 両方にマッチする。 

187 """ 

188 if fnmatch.fnmatch(normalized, pattern): 

189 return True 

190 # "**/.env" のようなパターンはルート直下（"/" なし）にも適用する 

191 if pattern.startswith("**/"): 

192 if fnmatch.fnmatch(normalized, pattern[3:]): 

193 return True 

194 return False

197def _match_path_patterns(patterns: List[str], paths: List[str], cwd: str) -> List[str]: 

198 """glob パターンにマッチしたパスのリストを返す。""" 

199 matched = [] 

200 for path_str in paths: 

201 normalized = _to_posix_relative(path_str, cwd) 

202 for pattern in patterns: 

203 if _match_single_path(normalized, pattern): 

204 matched.append(path_str) 

205 break 

206 return matched

209def _literal_prefix(pattern: str) -> str: 

210 """Return the literal path prefix before any wildcard segment.""" 

211 parts = [part for part in pattern.split("/") if part] 

212 prefix_parts: List[str] = [] 

213 for part in parts: 

214 if any(token in part for token in ("*", "?", "[")): 

215 break 

216 prefix_parts.append(part) 

217 return "/".join(prefix_parts)

220def _has_extra_constraints(pattern: str, prefix: str) -> bool: 

221 """Return True when pattern constrains matches beyond a plain recursive subtree.""" 

222 if not prefix: 

223 return pattern not in ("**", "**/*", "*") 

224 recursive_forms = {f"{prefix}/**", f"{prefix}/**/*", prefix} 

225 return pattern not in recursive_forms

228def _match_command_patterns(patterns: List[str], command: str, debug: Optional[HookDebugLogger] = None) -> List[str]: 

229 """Command patterns are matched as regexes with IGNORECASE flag.""" 

230 matched = [] 

231 for pattern in patterns: 

232 try: 

233 if re.search(pattern, command, re.IGNORECASE): 

234 matched.append(pattern) 

235 except re.error as exc: 

236 if debug: 

237 debug.log("regex_error", log_type="regex_error", pattern=pattern, error=str(exc)) 

238 return matched

241def _classify_overlap(request: str, rule: str) -> OverlapKind: 

242 """Classify overlap relationship between glob patterns 'request' and 'rule'.""" 

243 if not request or not rule: 

244 return OverlapKind.UNCERTAIN_OVERLAP 

245 

246 if rule.startswith("**/") and request == rule[3:]: 

247 return OverlapKind.DEFINITE_INCLUSION 

248 

249 # request が全カバー → request が rule を確実に包含 

250 if request in ("**", "**/*", "*"): 

251 return OverlapKind.DEFINITE_INCLUSION 

252 

253 request_prefix = _literal_prefix(request) 

254 rule_prefix = _literal_prefix(rule) 

255 

256 if request.endswith("/**") or request.endswith("/**/*"): 

257 if request_prefix and rule_prefix: 

258 if rule_prefix == request_prefix or rule_prefix.startswith(request_prefix + "/"): 

259 if _has_extra_constraints(rule, rule_prefix): 

260 return OverlapKind.UNCERTAIN_OVERLAP 

261 return OverlapKind.DEFINITE_INCLUSION 

262 

263 # 明確に非交差: ルートレベルが異なり、どちらも ** で始まらない 

264 request_root = request.split("/")[0] 

265 rule_root = rule.split("/")[0] 

266 if request_root != "**" and rule_root != "**" and request_root != rule_root: 

267 return OverlapKind.DEFINITE_DISJOINT 

268 

269 # その他は不確実交差 

270 return OverlapKind.UNCERTAIN_OVERLAP

273def _match_scope_patterns(request: str, patterns: List[str]) -> Optional[OverlapKind]: 

274 """Return overlap classification for a scope-like request against rule patterns. 

275 

276 Returns: 

277 OverlapKind.DEFINITE_INCLUSION if request certainly includes a protected scope, 

278 OverlapKind.UNCERTAIN_OVERLAP if overlap cannot be ruled out, 

279 None if all patterns are definitely disjoint. 

280 """ 

281 if not request or not patterns: 

282 return None 

283 

284 has_uncertain = False 

285 for pattern in patterns: 

286 overlap = _classify_overlap(request, pattern) 

287 if overlap == OverlapKind.DEFINITE_INCLUSION: 

288 return OverlapKind.DEFINITE_INCLUSION 

289 if overlap == OverlapKind.UNCERTAIN_OVERLAP: 

290 has_uncertain = True 

291 

292 if has_uncertain: 

293 return OverlapKind.UNCERTAIN_OVERLAP 

294 return None

297def _matches_rule( 

298 rule: Rule, 

299 context: MatchContext, 

300 operation_type: OperationType, 

301 debug: Optional[HookDebugLogger] = None, 

302) -> Optional[RuleMatch]: 

303 """ルールがコンテキストにマッチするか判定する。マッチすれば RuleMatch を返す。 

304 

305 拡張ガイド: 

306 新しい when フィールド（agent_id 等）は、ここに対応する条件チェックを追加する。 

307 各チェックは独立したブロックとして実装し、AND 条件で評価することを推奨する。 

308 """ 

309 when = rule.when 

310 

311 if operation_type in (OperationType.WRITE, OperationType.READ): 

312 paths = _get_paths_from_tool_input(context.tool_name, context.tool_input) 

313 

314 if operation_type == OperationType.WRITE: 

315 if when.path_patterns: 

316 matched = _match_path_patterns(when.path_patterns, paths, context.cwd) 

317 if matched: 

318 return RuleMatch(rule=rule, matched_values=matched) 

319 return None 

320 return RuleMatch(rule=rule, matched_values=paths) 

321 

322 concrete_paths: List[str] = [] 

323 scope_values: List[str] = [] 

324 ambiguous_values: List[str] = [] 

325 for path_str in paths: 

326 input_kind = classify_read_input(context.tool_name, path_str) 

327 if input_kind == "concrete_paths": 

328 concrete_paths.append(path_str) 

329 elif input_kind == "scope_patterns": 

330 scope_values.append(path_str) 

331 else: 

332 ambiguous_values.append(path_str) 

333 

334 if when.path_patterns: 

335 matched = _match_path_patterns(when.path_patterns, concrete_paths, context.cwd) 

336 if matched: 

337 return RuleMatch(rule=rule, matched_values=matched) 

338 

339 scope_patterns = when.scope_patterns or when.path_patterns 

340 if scope_patterns: 

341 for scope_value in scope_values: 

342 overlap = _match_scope_patterns(scope_value, scope_patterns) 

343 if overlap == OverlapKind.DEFINITE_INCLUSION: 

344 return RuleMatch(rule=rule, matched_values=[scope_value]) 

345 if overlap == OverlapKind.UNCERTAIN_OVERLAP and rule.action == "confirm": 

346 return RuleMatch(rule=rule, matched_values=[scope_value]) 

347 

348 if not paths and context.tool_name in ("grep_search", "file_search"): 

349 raw_scope_value = "" 

350 if context.tool_name == "grep_search": 

351 raw_scope_value = context.tool_input.get("includePattern", "") 

352 if context.tool_name == "file_search": 

353 raw_scope_value = context.tool_input.get("query", "") 

354 if raw_scope_value == "" and rule.action == "confirm": 

355 return RuleMatch(rule=rule, matched_values=[raw_scope_value]) 

356 

357 if ambiguous_values and rule.action == "confirm": 

358 return RuleMatch(rule=rule, matched_values=ambiguous_values) 

359 

360 if not when.path_patterns and not when.scope_patterns: 

361 return RuleMatch(rule=rule, matched_values=paths) 

362 

363 return None 

364 

365 if operation_type == OperationType.COMMAND: 

366 command = _get_command_from_tool_input(context.tool_input) 

367 if when.command_patterns: 

368 matched_patterns = _match_command_patterns(when.command_patterns, command, debug=debug) 

369 if matched_patterns: 

370 return RuleMatch(rule=rule, matched_values=matched_patterns) 

371 return None 

372 # command_patterns 未指定 → 全コマンドにマッチ 

373 return RuleMatch(rule=rule, matched_values=[command] if command else []) 

374 

375 return None

408def _get_candidate_rules(config: AccessControlConfig, operation_type: OperationType) -> List[Rule]: 

409 """操作タイプに対応するグループが有効な場合そのルールリストを、無効な場合空リストを返す。""" 

410 group = config.get_group(operation_type.value) 

411 return group.rules if group.enabled else []

414def _select_whitelist_arbitration( 

415 config: AccessControlConfig, operation_type: OperationType, context: MatchContext, debug: Optional[HookDebugLogger] = None 

416) -> Optional[str]: 

417 whitelist = config.whitelist 

418 if not whitelist.enabled: 

419 return None 

420 

421 group = whitelist.get_group(operation_type.value) 

422 if not group.enabled: 

423 return None 

424 

425 matched_arbitrations: List[str] = [] 

426 for rule in group.rules: 

427 if not rule.is_active(): 

428 continue 

429 match = _matches_whitelist_rule(rule, context, operation_type, debug=debug) 

430 if match: 

431 arbitration = rule.arbitration 

432 if arbitration in VALID_ARBITRATIONS: 

433 matched_arbitrations.append(arbitration) 

434 

435 for arbitration in _ARBITRATION_PRIORITY: 

436 if arbitration in matched_arbitrations: 

437 return arbitration 

438 

439 return None

442def _apply_arbitration(black_match: RuleMatch, arbitration: str) -> Optional[RuleMatch]: 

443 if arbitration == "blacklist": 

444 return black_match 

445 if arbitration == "confirm": 

446 return RuleMatch( 

447 rule=replace(black_match.rule, action="confirm"), 

448 matched_values=black_match.matched_values, 

449 ) 

450 if arbitration == "whitelist": 

451 return None 

452 return black_match

459def evaluate(config: AccessControlConfig, context: MatchContext, debug: Optional[HookDebugLogger] = None) -> Optional[RuleMatch]: 

460 """適用すべきルールを評価し、優先度が最も高いマッチを返す。 

461 

462 評価順序: 

463 1. context.tool_name から操作タイプを決定する 

464 2. 対応するルールリストを取得する 

465 3. action="disabled" のルールを除外する 

466 4. 残りのルールをコンテキストと照合する 

467 5. マッチしたルールの中から deny > confirm の優先度で1件返す 

468 6. マッチなし = None (allow) 

469 

470 Args: 

471 config: 読み込み済みのアクセス制御設定。 

472 context: 現在のツール呼び出しコンテキスト。 

473 

474 Returns: 

475 最も優先度の高い RuleMatch、またはマッチなしの場合は None。 

476 """ 

477 operation_type = _determine_operation_type(context.tool_name) 

478 if operation_type is None: 

479 return None 

480 

481 if not config.enabled: 

482 return None 

483 

484 candidates = _get_candidate_rules(config, operation_type) 

485 

486 active_matches: List[RuleMatch] = [] 

487 for rule in candidates: 

488 if not rule.is_active(): 

489 continue 

490 match = _matches_rule(rule, context, operation_type, debug=debug) 

491 if match: 

492 active_matches.append(match) 

493 

494 if not active_matches: 

495 return None 

496 

497 # 優先度解決: deny > confirm 

498 for action in _ACTION_PRIORITY: 

499 for match in active_matches: 

500 if match.rule.action == action: 

501 arbitration = _select_whitelist_arbitration( 

502 config=config, 

503 operation_type=operation_type, 

504 context=context, 

505 debug=debug, 

506 ) 

507 if arbitration is None: 

508 return match 

509 return _apply_arbitration(match, arbitration) 

510 

511 return None
