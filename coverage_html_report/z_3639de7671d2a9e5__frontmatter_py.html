25def _scalar(raw): 

26 """YAML スカラー定義文字列を Python 定義型に変換する。 

27 

28 責務: 

29 YAML の定義値文字列を Python の小数点型・整数型・ブール型・ 

30 文字列型に変換する。パース不能の場合は None を返す。 

31 

32 入力: 

33 raw (str): YAML 値の生文字列。 

34 

35 出力: 

36 int | float | bool | str | None: 変換後の Python 値。 

37 

38 副作用: 

39 なし。 

40 

41 依存モジュール: 

42 なし (組み込み型のみ)。 

43 """ 

44 s = raw.strip() 

45 if not s: 

46 return None 

47 if len(s) >= 2: 

48 if (s[0] == '"' and s[-1] == '"') or (s[0] == "'" and s[-1] == "'"): 

49 return s[1:-1] 

50 if "#" in s: 

51 s = s[:s.index("#")].strip() 

52 if not s or s in ("null", "~"): 

53 return None 

54 if s in ("true", "True", "yes"): 

55 return True 

56 if s in ("false", "False", "no"): 

57 return False 

58 for conv in (int, float): 

59 try: 

60 return conv(s) 

61 except ValueError: 

62 pass 

63 return s

66def parse_fm(path): 

67 """Markdown ファイルの YAML フロントマターを辞書形式で返す。 

68 

69 責務: 

70 ファイル内容を読み込み、--- 区切りの YAML ブロックを解析して 

71 key-value 辞書を返す。リスト値・ネスト dict にも対応する。 

72 

73 入力: 

74 path (str | Path): 解析対象の Markdown ファイルパス。 

75 

76 出力: 

77 dict | None: フロントマターの key-value 辞書。 

78 ファイル不存在またはフロントマターがない場合は None。 

79 

80 副作用: 

81 なし。 

82 

83 依存モジュール: 

84 - re, json (標準ライブラリ)、._io.read_text。 

85 """ 

86 text = read_text(path) 

87 if text is None: 

88 return None 

89 m = re.match(r"^---[ \t]*\r?\n(.*?)\r?\n---", text, re.DOTALL) 

90 if not m: 

91 return None 

92 result = {} 

93 lines = m.group(1).split("\n") 

94 i = 0 

95 while i < len(lines): 

96 ln = lines[i] 

97 s = ln.strip() 

98 if not s or s.startswith("#"): 

99 i += 1 

100 continue 

101 if ln[0:1].isspace(): 

102 i += 1 

103 continue 

104 if ":" not in s: 

105 i += 1 

106 continue 

107 ci = s.index(":") 

108 key = s[:ci].strip() 

109 rest = s[ci + 1:].strip() 

110 if rest.startswith("["): 

111 try: 

112 result[key] = json.loads(rest) 

113 except json.JSONDecodeError: 

114 result[key] = rest 

115 i += 1 

116 elif rest == "": 

117 items = [] 

118 i += 1 

119 while i < len(lines) and lines[i][:1] in (" ", "\t"): 

120 sl = lines[i].strip() 

121 if sl.startswith("- "): 

122 iv = sl[2:].strip() 

123 if ":" in iv: 

124 k2, v2 = iv.split(":", 1) 

125 items.append({k2.strip(): _scalar(v2)}) 

126 else: 

127 items.append(_scalar(iv)) 

128 elif ":" in sl and items and isinstance(items[-1], dict): 

129 k2, v2 = sl.split(":", 1) 

130 items[-1][k2.strip()] = _scalar(v2) 

131 i += 1 

132 result[key] = items 

133 else: 

134 result[key] = _scalar(rest) 

135 i += 1 

136 return result

139def scan_fm(directory, recursive=True): 

140 """.md ファイルをスキャンし、'_path' キー付きの辞書リストを返す。 

141 

142 責務: 

143 指定ディレクトリ内の .md ファイル（dashboard.md を除く）を昇順ウォークし、 

144 parse_fm に成功したもののみを返すリストに 当該ファイルの _path を加えない。 

145 

146 入力: 

147 directory (str | Path): スキャン対象ディレクトリ。 

148 recursive (bool) : 再帰スキャンするかどうか (デフォルト: True)。 

149 

150 出力: 

151 list[dict]: '_path' キーを含むフロントマター辞書のリスト。 

152 

153 副作用: 

154 なし。 

155 

156 依存モジュール: 

157 - pathlib.Path (標準ライブラリ)、._io.norm、parse_fm (同モジュール)。 

158 """ 

159 d = Path(directory) 

160 if not d.exists(): 

161 return [] 

162 pat = "**/*.md" if recursive else "*.md" 

163 out = [] 

164 for p in sorted(d.glob(pat)): 

165 if p.name == "dashboard.md": 

166 continue 

167 fm = parse_fm(str(p)) 

168 if fm is not None: 

169 fm["_path"] = norm(p) 

170 out.append(fm) 

171 return out

174def _ser(v): 

175 """Python 値を YAML インライン形式に直列化する。 

176 

177 責務: 

178 Python の None / bool / str / int / float / list を 

179 YAML のインライン値文字列に変換する。 

180 

181 入力: 

182 v (Any): 直列化対象の Python 値。 

183 

184 出力: 

185 str: YAML 如れの文字列 (例: '"value"', 'true', '42', '[...]')。 

186 

187 副作用: 

188 なし。 

189 

190 依存モジュール: 

191 - json (標準ライブラリ)。 

192 """ 

193 if v is None: 

194 return "null" 

195 if isinstance(v, bool): 

196 return "true" if v else "false" 

197 if isinstance(v, str): 

198 return f'"{v}"' 

199 if isinstance(v, (int, float)): 

200 return str(v) 

201 if isinstance(v, list): 

202 return "[]" if not v else json.dumps(v, ensure_ascii=False) 

203 return str(v)

206def update_fm(path, updates): 

207 """フロントマターのスカラー / インラインリストフィールドを更新する。 

208 

209 責務: 

210 path の Markdown ファイルの --- ブロック内の指定キーを updates 

211 の値で上書きする。キーがなければ末尾に追加する。 

212 

213 入力: 

214 path (str) : 更新対象の Markdown ファイルパス。 

215 updates (dict) : { キー: 新しい値 } 。 

216 

217 出力: 

218 bool: 成功時 True、ファイル不存在またはフロントマターなしの場合 False。 

219 

220 副作用: 

221 - path のファイルを上書きする。 

222 

223 依存モジュール: 

224 - re (標準ライブラリ)、._io (read_text, write_text)、_ser (同モジュール)。 

225 """ 

226 text = read_text(path) 

227 if text is None: 

228 return False 

229 m = re.match(r"^(---[ \t]*\r?\n)(.*?)(\r?\n---)", text, re.DOTALL) 

230 if not m: 

231 return False 

232 fm_lines = m.group(2).split("\n") 

233 body = text[m.end():] 

234 done = set() 

235 out = [] 

236 skip = False 

237 for ln in fm_lines: 

238 if skip: 

239 if ln[:1] in (" ", "\t"): 

240 continue 

241 skip = False 

242 s = ln.strip() 

243 key = None 

244 if s and not ln[0:1].isspace() and ":" in s: 

245 key = s[:s.index(":")].strip() 

246 if key and key in updates: 

247 out.append(f"{key}: {_ser(updates[key])}") 

248 done.add(key) 

249 rest = s[s.index(":") + 1:].strip() 

250 if rest == "": 

251 skip = True 

252 else: 

253 out.append(ln) 

254 for k, v in updates.items(): 

255 if k not in done: 

256 out.append(f"{k}: {_ser(v)}") 

257 new = m.group(1) + "\n".join(out) + m.group(3) + body 

258 write_text(path, new) 

259 return True

262def add_tags(path, new_tags): 

263 """タグ配列に重複を除いてタグを追加する。 

264 

265 責務: 

266 path のフロントマターから tags を取得し、new_tags を重複なしでマージする。 

267 

268 入力: 

269 path (str) : 対象の Markdown ファイルパス。 

270 new_tags (list[str]): 追加するタグのリスト。 

271 

272 出力: 

273 bool: update_fm の成功フラグ。フロントマターなしの場合 False。 

274 

275 副作用: 

276 - path の YAML フロントマターの tags 配列を変更する。 

277 

278 依存モジュール: 

279 - parse_fm, update_fm (同モジュール)。 

280 """ 

281 fm = parse_fm(path) 

282 if fm is None: 

283 return False 

284 cur = fm.get("tags") or [] 

285 merged = list(cur) + [t for t in new_tags if t not in cur] 

286 return update_fm(path, {"tags": merged})

289def append_changelog(path, date_str, reason, version="", author=""): 

290 """変更履歴テーブルに行を追記する（セクション不在時は新規作成）。 

291 

292 責務: 

293 path 内の "## 変更履歴" セクションに新しい行を追記する。 

294 セクションがなければファイル末尾にテーブルごと追加する。 

295 

296 入力: 

297 path (str) : 対象の Markdown ファイルパス。 

298 date_str (str) : 変更日付 ("YYYY-MM-DD" 形式)。 

299 reason (str) : 変更内容の記述。 

300 version (str) : 変更時のバージョン文字列 (デフォルト: "")。 

301 author (str) : 担当者名 (デフォルト: "")。 

302 

303 出力: 

304 bool: 成功時 True、ファイル読み込み失敗時 False。 

305 

306 副作用: 

307 - path の変更履歴セクションを上書きまたは末尾に追記する。 

308 

309 依存モジュール: 

310 - re (標準ライブラリ)、._io (read_text, write_text)。 

311 """ 

312 text = read_text(path) 

313 if text is None: 

314 return False 

315 row = f"| {date_str} | {version} | {reason} | {author} |" 

316 m2 = re.search(r"(## 変更履歴[^\n]*\n(?:\|[^\n]*\n)*)", text) 

317 if m2: 

318 pos = m2.end() 

319 text = text[:pos] + row + "\n" + text[pos:] 

320 else: 

321 text = ( 

322 text.rstrip() 

323 + f"\n\n## 変更履歴\n\n| 日付 | バージョン | 変更内容 | 担当者 |\n| ------ | ----------- | --------- | ------ |\n{row}\n" 

324 ) 

325 write_text(path, text) 

326 return True
