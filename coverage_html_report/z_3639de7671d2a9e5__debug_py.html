1#!/usr/bin/env python3 

2"""Per-script debug flag and logging. 

3 

4責務: 

5 各スクリプト固有のデバッグフラグファイルの存在をチェックし、 

6 有効時のみ .debug.log ファイルにタイムスタンプ付きログを追記する。 

7 各スクリプトは個別のデバッグフラグ / ログファイルを持つ。 

8 フラグ: <script_key>.debug (存在で有効化) 

9 ログ : <script_key>.debug.log (デバッグ有効時に自動生成) 

10 script_key: "R-1:resolve-cascade-scope" → "resolve_cascade_scope" 

11 

12入力 / 出力: 

13 各関数の docstring を参照。 

14 

15副作用: 

16 debug_log が .debug.log ファイルへ追記する。 

17 

18依存モジュール: 

19 os, json, datetime (はいずれも標準ライブラリ)。 

20""" 

21 

22import os, json 

23from datetime import datetime 

24 

25_SCRIPTS_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))

28def _script_key(script): 

29 """script 識別子からファイルセーフなキーを抽出する。 

30 

31 責務: 

32 "R-1:resolve-cascade-scope" のようなスクリプト識別子から 

33 ":" 以降の部分を取り出し、"-" を "_" に展開して返す。 

34 

35 入力: 

36 script (str): "<プレフィックス>:<名前>" 形式または単笔名前。 

37 

38 出力: 

39 str: ファイル名に使えるキー文字列。 

40 

41 副作用: 

42 なし。 

43 

44 依存モジュール: 

45 なし (組み込み str のみ)。 

46 """ 

47 s = script.split(":", 1)[-1] if ":" in script else script 

48 return s.replace("-", "_")

51def is_debug(script): 

52 """指定スクリプトのデバッグモードが有効かどうかを返す。 

53 

54 責務: 

55 <script_key>.debug ファイルの存在を確認する。 

56 

57 入力: 

58 script (str): スクリプト識別子 ("R-1:resolve-cascade-scope" 形式)。 

59 

60 出力: 

61 bool: デバッグフラグファイルが存在する場合 True。 

62 

63 副作用: 

64 なし。 

65 

66 依存モジュール: 

67 - os (標準ライブラリ)、_script_key (同モジュール)。 

68 """ 

69 key = _script_key(script) 

70 return os.path.exists(os.path.join(_SCRIPTS_DIR, f"{key}.debug"))

73def debug_log(script, msg, **kw): 

74 """タイムスタンプ付きデバッグログエントリをスクリプト固有のログファイルに追記する。 

75 

76 責務: 

77 <script_key>.debug ファイルが存在する場合のみ、 

78 [timestamp] [script] msg | {kw} 形式で .debug.log に追記する。 

79 

80 入力: 

81 script (str) : スクリプト識別子。 

82 msg (str) : ログメッセージ。 

83 **kw : ログに付加する任意のキーワード引数。 

84 

85 出力: 

86 なし (None)。 

87 

88 副作用: 

89 - デバッグモード時に <script_key>.debug.log へ行を追記する。 

90 

91 依存モジュール: 

92 - os, json, datetime (標準ライブラリ)、_script_key (同モジュール)。 

93 """ 

94 key = _script_key(script) 

95 if not os.path.exists(os.path.join(_SCRIPTS_DIR, f"{key}.debug")): 

96 return 

97 log_path = os.path.join(_SCRIPTS_DIR, f"{key}.debug.log") 

98 ts = datetime.now().strftime("%Y-%m-%d %H:%M:%S") 

99 line = f"[{ts}] [{script}] {msg}" 

100 if kw: 

101 line += " | " + json.dumps(kw, ensure_ascii=False, default=str) 

102 with open(log_path, "a", encoding="utf-8") as f: 

103 f.write(line + "\n")
